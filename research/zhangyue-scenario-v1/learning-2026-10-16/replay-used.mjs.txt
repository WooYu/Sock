function match(price, actual) {
  const predicted = Math.round(price * 100), tick = Math.round(actual * 100)
  // Integer comparison enforces strict <1%, including the exact boundary.
  return { relativeError: Math.abs(predicted - tick) / tick, close: Math.abs(predicted - tick) * 100 < tick, exact: predicted === tick }
}
function evaluate(levels, actual) {
  const details = levels.map(level => {
    const fields = Object.fromEntries(protocol.matching.fields.map(field => [field, match(level.value, actual[field])]))
    return { price: level.value, groupIds: level.groupIds, fields,
      close: Object.values(fields).some(value => value.close), exact: Object.values(fields).some(value => value.exact) }
  })
  const targets = protocol.matching.fields.map(field => {
    const nearest = details.toSorted((a, b) => a.fields[field].relativeError - b.fields[field].relativeError)[0]
    return { field, actual: actual[field], nearestPrice: nearest?.price ?? null,
      relativeError: nearest?.fields[field].relativeError ?? null,
      close: nearest?.fields[field].close ?? false, exact: nearest?.fields[field].exact ?? false }
  })
  const closeCount = details.filter(row => row.close).length
  const targetCount = targets.filter(row => row.close).length
  return { levels: details, targets, count: details.length, closeCount, exactCount: details.filter(row => row.exact).length,
    precision: details.length ? closeCount / details.length : 0, coverage: targetCount / 4,
    targetCount, empty: details.length === 0 }
}
function summary(records, key) {
  const rows = records.map(record => record.methods[key])
  const sum = mapper => rows.reduce((total, row) => total + mapper(row), 0)
  const p = sum(row => row.precision) / rows.length, c = sum(row => row.coverage) / rows.length
  return { samples: rows.length, meanDailyPrecision: p, meanDailyCoverage: c, selectionScore: p + c ? 2 * p * c / (p + c) : 0,
    totalLevels: sum(row => row.count), closeLevels: sum(row => row.closeCount), exactLevels: sum(row => row.exactCount),
    microPrecision: sum(row => row.count) ? sum(row => row.closeCount) / sum(row => row.count) : null,
    meanCandidates: sum(row => row.count) / rows.length, emptyDays: rows.filter(row => row.empty).length,
    targetCoverage: Object.fromEntries(protocol.matching.fields.map(field => [field, sum(row => Number(row.targets.find(target => target.field === field).close)) / rows.length])),
    meanNearestTargetRelativeError: Object.fromEntries(protocol.matching.fields.map(field => {
      const values = rows.map(row => row.targets.find(target => target.field === field).relativeError).filter(value => value !== null)
      return [field, values.length ? values.reduce((a, b) => a + b, 0) / values.length : null]
    })),
  }
}
